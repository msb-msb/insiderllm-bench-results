#!/bin/bash
set -euo pipefail
cd ~
echo "start $(date -Is)"
git clone --depth 1 --branch v0.4.0 https://github.com/ggml-org/llama.cpp ~/llama-v0.4.0
cd ~/llama-v0.4.0
git log -1 --format="%H %d %s"
cmake -B build -DGGML_CUDA=ON -DCMAKE_CUDA_ARCHITECTURES=86 -DGGML_NATIVE=OFF -DCMAKE_BUILD_TYPE=Release -DLLAMA_CURL=OFF \
  -DCMAKE_C_COMPILER=gcc-13 -DCMAKE_CXX_COMPILER=g++-13 -DCMAKE_CUDA_HOST_COMPILER=g++-13
cmake --build build --config Release -j 16
echo "build done $(date -Is)"
CMAKE_BUILD_TYPE:STRING=Release
CMAKE_CUDA_ARCHITECTURES:UNINITIALIZED=86
CMAKE_CUDA_COMPILER:FILEPATH=/usr/bin/nvcc
CMAKE_CUDA_HOST_COMPILER:UNINITIALIZED=/usr/bin/g++-13
CMAKE_CXX_COMPILER:STRING=/usr/bin/g++-13
CMAKE_C_COMPILER:STRING=/usr/bin/gcc-13
GGML_AVX2:BOOL=ON
GGML_AVX512:BOOL=OFF
GGML_CUDA:BOOL=ON
GGML_CUDA_COMPRESSION_MODE:STRING=size
GGML_CUDA_FA:BOOL=ON
GGML_CUDA_FORCE_CUBLAS:BOOL=OFF
GGML_CUDA_FORCE_MMQ:BOOL=OFF
GGML_CUDA_GRAPHS:BOOL=ON
GGML_CUDA_NCCL:BOOL=ON
GGML_NATIVE:BOOL=OFF
GGML_OPENMP:BOOL=ON
LLAMA_CURL:UNINITIALIZED=OFF
CMAKE_CUDA_HOST_COMPILER-ADVANCED:INTERNAL=1
GGML_CUDA_COMPRESSION_MODE-STRINGS:INTERNAL=none;speed;balance;size
